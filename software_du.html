def elastischer_stoss(m1, m2, v1, v2):
  v1_f = ((m1 - m2)*v1 + 2*m2*v2) / (m1 + m2)
  v2_f = ((m2 - m1)*v2 + 2*m1*v1) / (m1 + m2)
  return v1_f, v2_f
