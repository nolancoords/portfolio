def elastic_collision(m1, m2, v1, v2):
  v1_f = ((m1 - m2)*v1 + 2*m2*v2) / (m1 + m2)
  v2_f = ((m2 - m1)*v2 + 2*m1*v1) / (m1 + m2)
  return v1_f, v2_f

double Beta(const vector<double>& s, const vector<double>& m) {
  if (s.size() != m.size() || s.empty()) return 0;
  double sm = accumulate(s.begin(), s.end(), 0.0) / s.size();
  double mm = accumulate(m.begin(), m.end(), 0.0) / m.size();
  double cov = 0, mvar = 0;
  for (size_t i = 0; i < s.size(); ++i) {
    cov += (s[i] - sm) * (m[i] - mm);
    mvar += (m[i] - mm) * (m[i] - mm);
  }
  return mvar != 0 ? cov / mvar : 0;
}
